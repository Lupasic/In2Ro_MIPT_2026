\begin{frame}[t]{What this homework is}
\framesubtitle{One arm, six tasks, eighteen hours}
    AR-601M is a 41-DOF humanoid robot. This homework treats one of its two arms in
    isolation: a 6-DOF serial chain, shoulder to tool, with a spherical wrist.
    \medskip

    \begin{itemize}
        \item \textbf{Forward} kinematics: joint angles $\to$ tool pose.
        \item \textbf{Inverse} kinematics, analytical and numerical: tool pose $\to$ joint
              angles.
        \item \textbf{Differential kinematics} and \textbf{singularities}.
    \end{itemize}
    \medskip

    About 18 hours of work covering weeks 1--3. This deck is a pointer; the full statement,
    with every derivation, requirement and hint, is the handout linked on the last frame.
\end{frame}

\begin{frame}[t]{The six tasks}
    \begin{multicols}{2}
        \begin{itemize}
            \item \textbf{1.} Kinematic model
            \item \textbf{2.} FK: angles $\to$ arm and pose
            \item \textbf{3.} Analytical IK: pose $\to$ all feasible branches
            \item \textbf{4.} Numerical IK: solve from multiple seeds
            \item \textbf{5.} Differential kinematics
            \item \textbf{6.} Singularity and solution-count maps
        \end{itemize}
    \end{multicols}
    \medskip

    Full requirements and evidence for every task are in handout \S4.
\end{frame}

\begin{frame}[t]{Required visualizations}
    \begin{enumerate}
        \item \textbf{FK:} enter six joint angles; draw the manipulator and print
              the resulting homogeneous matrix $T$.
        \item \textbf{Analytical IK:} enter a pose from FK or a position and
              ZYX Euler orientation; draw \emph{every distinct feasible}
              configuration and print each complete six-angle vector.
        \item \textbf{Singularities:} plot singularity information and a
              fixed-orientation workspace slice. Use a continuous color
              scale for the number of feasible IK configurations, from
              zero to the maximum attained in this case.
    \end{enumerate}
    Mark unresolved targets and continuous families separately from finite counts.
\end{frame}
